\begin{frame}{O Estilo Microkernel}
\begin{itemize}\setlength{\itemsep}{0.65em}
\item Também conhecido como \textbf{Arquitetura Baseada em Plugins}.
\item Resolve a necessidade de adicionar novas funcionalidades sem inflar ou quebrar o núcleo do sistema (evitando o antipadrão \textit{Big Ball of Mud}).
\item \textbf{Estrutura Central:} Um \textit{Core System} enxuto + múltiplos \textit{Plug-in Components} independentes.
\item \textbf{Objetivo do trabalho:} Analisar e documentar este estilo em dois sistemas reais e \textit{open source}: \textbf{Visual Studio Code} e \textbf{OBS Studio}.
\end{itemize}
\fonte{Richards e Ford (2020); elaboração própria}
\end{frame}

\begin{frame}{Visual Studio Code: Decisões Arquiteturais (ADRs)}
\begin{itemize}\setlength{\itemsep}{0.65em}
\item \textbf{ADR-001: Adoção do Estilo Microkernel}
  \begin{itemize} \item Desacoplamento por meio de um vocabulário claro de componentes. \end{itemize}
\item \textbf{ADR-002: Isolamento de Extensões via Extension Host}
  \begin{itemize} \item Executa plugins em processos separados. Evita travamentos na interface, em troca de um ligeiro custo de comunicação remota. \end{itemize}
\item \textbf{ADR-003: Implementação do Language Server Protocol (LSP)}
  \begin{itemize} \item Particionamento de domínio. Os serviços de sintaxe rodam como processos distribuídos. \end{itemize}
\item \textbf{ADR-004: Utilização do Framework Electron}
  \begin{itemize} \item Encapsulamento Web (HTML/JS) visando portabilidade extrema, assumindo um \textit{trade-off} de consumo de recursos. \end{itemize}
\end{itemize}

\textbf{Característica} & \textbf{Justificativa do Foco}\\\midrule
Extensibilidade & Módulos (fontes, codificadores, saídas) adicionados via registro padronizado sem alterar o núcleo (\texttt{libobs}).\\[0.3em]
Desempenho & Processamento multimídia a 60 FPS exige que a latência de compressão e cópia seja quase nula.\\[0.3em]
Confiabilidade e Tol. a Falhas & Uma falha de hardware, como a queda da rede, não pode congelar ou encerrar o loop principal de vídeo.\\[0.3em]
Implantabilidade/Portabilidade & Distribuição consistente (Windows, macOS, Linux) e capacidade de injetar plugins dinamicamente.\\\bottomrule
\end{tabularx}

\begin{frame}{OBS Studio: Decisões Arquiteturais (ADRs)}
\begin{itemize}\setlength{\itemsep}{0.65em}
\item \textbf{ADR-IMP-001: Implementação do Núcleo em C}
  \begin{itemize} \item O \textit{libobs} é escrito em C para maximizar velocidade, controle de memória e possuir uma Interface Binária (ABI) estável. \end{itemize}
\item \textbf{ADR-IMP-002: Desacoplamento da Interface Visual por meio de Qt}
  \begin{itemize} \item Frontend escrito em C++ com framework Qt, separado do motor. \end{itemize}
\item \textbf{ADR-IMP-003: Compartilhamento de Contexto Gráfico (Zero-Copy)}
  \begin{itemize} \item Plugins repassam referências na GPU para evitar a cópia massiva de pixels na CPU/RAM. \end{itemize}
\item \textbf{ADR-IMP-004: Injeção de Funcionalidades via Carregamento Dinâmico}
  \begin{itemize} \item Uso de bibliotecas dinâmicas do SO (.dll/.so) para carregar módulos sob demanda ao iniciar. \end{itemize}
\end{itemize}

\begin{frame}{Conclusão}
\begin{itemize}\setlength{\itemsep}{0.8em}
\item O estilo \textbf{Microkernel} isola a orquestração central das funcionalidades específicas, sendo ideal para ecossistemas que evoluem continuamente.
\item \textbf{VS Code:} Focou no isolamento rigoroso de processos (\textit{Extension Host}) para garantir que a interface do usuário sobreviva a códigos defeituosos.
\item \textbf{OBS Studio:} Focou no desempenho nativo em C e no compartilhamento de memória via GPU para garantir latência nula de vídeo.
\item \textbf{Aprendizado Crítico:} A definição de contratos (APIs) bem estruturados no \textit{Core System} é o pilar que evita a degradação de arquiteturas em longo prazo.
\end{itemize}
\end{frame}
